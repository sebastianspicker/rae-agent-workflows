cmake_minimum_required(VERSION 3.15)
project(rae_fs_bridge C)
if(NOT APPLE AND NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "RAE requires descriptor-relative no-follow operations and atomic no-clobber rename")
endif()
add_library(rae_fs_bridge SHARED native/bridge.c ${CMAKE_JS_SRC})
set_target_properties(rae_fs_bridge PROPERTIES PREFIX "" SUFFIX ".node" C_STANDARD 11)
target_include_directories(rae_fs_bridge PRIVATE ${CMAKE_JS_INC})
target_link_libraries(rae_fs_bridge ${CMAKE_JS_LIB})
target_compile_definitions(rae_fs_bridge PRIVATE NAPI_VERSION=8 _GNU_SOURCE)
target_compile_options(rae_fs_bridge PRIVATE -Wall -Wextra -Werror)
